#include "W25Q64.h"

#define W25Q64_SPI              SPI1

#define W25Q64_CS_GPIO          GPIOA
#define W25Q64_CS_PIN           GPIO_Pin_4

#define W25Q64_CS_LOW()         GPIO_ResetBits(W25Q64_CS_GPIO, W25Q64_CS_PIN)
#define W25Q64_CS_HIGH()        GPIO_SetBits(W25Q64_CS_GPIO, W25Q64_CS_PIN)

/* W25Q64 commands */
#define W25Q64_CMD_WRITE_ENABLE     0x06
#define W25Q64_CMD_READ_STATUS1     0x05
#define W25Q64_CMD_PAGE_PROGRAM     0x02
#define W25Q64_CMD_READ_DATA        0x03
#define W25Q64_CMD_SECTOR_ERASE     0x20
#define W25Q64_CMD_JEDEC_ID         0x9F

#define W25Q64_STATUS_BUSY          0x01

static uint8_t W25Q64_SPI_SwapByte(uint8_t byte)
{
    while (SPI_I2S_GetFlagStatus(W25Q64_SPI, SPI_I2S_FLAG_TXE) == RESET);

    SPI_I2S_SendData(W25Q64_SPI, byte);

    while (SPI_I2S_GetFlagStatus(W25Q64_SPI, SPI_I2S_FLAG_RXNE) == RESET);

    return SPI_I2S_ReceiveData(W25Q64_SPI);
}

static void W25Q64_SendAddress(uint32_t addr)
{
    W25Q64_SPI_SwapByte((addr >> 16) & 0xFF);
    W25Q64_SPI_SwapByte((addr >> 8) & 0xFF);
    W25Q64_SPI_SwapByte(addr & 0xFF);
}

static void W25Q64_WriteEnable(void)
{
    W25Q64_CS_LOW();

    W25Q64_SPI_SwapByte(W25Q64_CMD_WRITE_ENABLE);

    W25Q64_CS_HIGH();
}

static uint8_t W25Q64_ReadStatus1(void)
{
    uint8_t status;

    W25Q64_CS_LOW();

    W25Q64_SPI_SwapByte(W25Q64_CMD_READ_STATUS1);
    status = W25Q64_SPI_SwapByte(0xFF);

    W25Q64_CS_HIGH();

    return status;
}

static void W25Q64_WaitBusy(void)
{
    while (W25Q64_ReadStatus1() & W25Q64_STATUS_BUSY);
}

void W25Q64_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStructure;
    SPI_InitTypeDef SPI_InitStructure;

    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA | RCC_APB2Periph_SPI1, ENABLE);

    /* CS: PA4 */
    GPIO_InitStructure.GPIO_Pin = W25Q64_CS_PIN;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(W25Q64_CS_GPIO, &GPIO_InitStructure);

    W25Q64_CS_HIGH();

    /* SCK: PA5, MOSI: PA7 */
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_5 | GPIO_Pin_7;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_AF_PP;
    GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(GPIOA, &GPIO_InitStructure);

    /* MISO: PA6 */
    GPIO_InitStructure.GPIO_Pin = GPIO_Pin_6;
    GPIO_InitStructure.GPIO_Mode = GPIO_Mode_IN_FLOATING;
    GPIO_Init(GPIOA, &GPIO_InitStructure);

    SPI_InitStructure.SPI_Direction = SPI_Direction_2Lines_FullDuplex;
    SPI_InitStructure.SPI_Mode = SPI_Mode_Master;
    SPI_InitStructure.SPI_DataSize = SPI_DataSize_8b;

    /*
     * W25Q64 支持 SPI Mode 0 和 Mode 3。
     * 这里用 Mode 0: CPOL = 0, CPHA = 0。
     */
    SPI_InitStructure.SPI_CPOL = SPI_CPOL_Low;
    SPI_InitStructure.SPI_CPHA = SPI_CPHA_1Edge;

    SPI_InitStructure.SPI_NSS = SPI_NSS_Soft;

    /*
     * F103 SPI1 时钟来自 APB2。
     * 如果 APB2 = 72MHz，4 分频后 SPI = 18MHz。
     */
    SPI_InitStructure.SPI_BaudRatePrescaler = SPI_BaudRatePrescaler_4;

    SPI_InitStructure.SPI_FirstBit = SPI_FirstBit_MSB;
    SPI_InitStructure.SPI_CRCPolynomial = 7;

    SPI_Init(W25Q64_SPI, &SPI_InitStructure);

    SPI_Cmd(W25Q64_SPI, ENABLE);
}

uint32_t W25Q64_ReadID(void)
{
    uint32_t id = 0;

    W25Q64_CS_LOW();

    W25Q64_SPI_SwapByte(W25Q64_CMD_JEDEC_ID);

    id |= W25Q64_SPI_SwapByte(0xFF) << 16;
    id |= W25Q64_SPI_SwapByte(0xFF) << 8;
    id |= W25Q64_SPI_SwapByte(0xFF);

    W25Q64_CS_HIGH();

    return id;
}

void W25Q64_ReadData(uint32_t addr, uint8_t *buf, uint32_t len)
{
    uint32_t i;

    W25Q64_CS_LOW();

    W25Q64_SPI_SwapByte(W25Q64_CMD_READ_DATA);
    W25Q64_SendAddress(addr);

    for (i = 0; i < len; i++)
    {
        buf[i] = W25Q64_SPI_SwapByte(0xFF);
    }

    W25Q64_CS_HIGH();
}

void W25Q64_SectorErase(uint32_t addr)
{
    W25Q64_WriteEnable();

    W25Q64_CS_LOW();

    W25Q64_SPI_SwapByte(W25Q64_CMD_SECTOR_ERASE);
    W25Q64_SendAddress(addr);

    W25Q64_CS_HIGH();

    W25Q64_WaitBusy();
}

void W25Q64_PageProgram(uint32_t addr, const uint8_t *buf, uint16_t len)
{
    uint16_t i;

    if (len > 256)
    {
        len = 256;
    }

    W25Q64_WriteEnable();

    W25Q64_CS_LOW();

    W25Q64_SPI_SwapByte(W25Q64_CMD_PAGE_PROGRAM);
    W25Q64_SendAddress(addr);

    for (i = 0; i < len; i++)
    {
        W25Q64_SPI_SwapByte(buf[i]);
    }

    W25Q64_CS_HIGH();

    W25Q64_WaitBusy();
}

void W25Q64_WriteData(uint32_t addr, const uint8_t *buf, uint32_t len)
{
    uint32_t pageRemain;

    while (len > 0)
    {
        pageRemain = 256 - (addr % 256);

        if (pageRemain > len)
        {
            pageRemain = len;
        }

        W25Q64_PageProgram(addr, buf, pageRemain);

        addr += pageRemain;
        buf += pageRemain;
        len -= pageRemain;
    }
}
